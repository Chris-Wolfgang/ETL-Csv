Examples

public record Person
{
    [CsvColumn(Name = "first_name")]
    public string FirstName { get; init; }

    [CsvColumn(Name = "last_name")]
    public string LastName { get; init; }

    [CsvColumn(Name = "dob", Format = "yyyy-MM-dd", Optional = true, Default = "1970-01-01")]
    public DateTime DateOfBirth { get; init; }
}

Remarks

This attribute lets consumers describe their CSV layout without depending on any specific CSV parser. The CSV library translates these settings to the underlying parser at runtime.
